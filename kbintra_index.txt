    <script>
        // 세션 타이머 카운트다운
        let sessionSeconds = 3590;
        const timerElement = document.getElementById('sessionTimer');

        setInterval(() => {
            if (sessionSeconds > 0) {
                sessionSeconds--;
                const minutes = Math.floor(sessionSeconds / 60);
                const seconds = sessionSeconds % 60;
                timerElement.textContent = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
            }
        }, 1000);

        function extendSession() {
            sessionSeconds = 3600;
            alert('보안 세션 시간이 60분 연장되었습니다.');
        }

        // 근무 상태 업데이트
        function updateWorkStatus(status) {
            const indicator = document.getElementById('statusIndicator');
            if (status === 'working') {
                indicator.className = "absolute bottom-0 right-0 w-2.5 h-2.5 bg-emerald-500 border-2 border-white rounded-full";
            } else if (status === 'away') {
                indicator.className = "absolute bottom-0 right-0 w-2.5 h-2.5 bg-amber-500 border-2 border-white rounded-full";
            } else if (status === 'meal') {
                indicator.className = "absolute bottom-0 right-0 w-2.5 h-2.5 bg-orange-500 border-2 border-white rounded-full";
            } else {
                indicator.className = "absolute bottom-0 right-0 w-2.5 h-2.5 bg-red-500 border-2 border-white rounded-full";
            }
        }

        // 임직원 전화번호부 빠른 검색
        function filterEmployees() {
            const input = document.getElementById('employeeSearchInput').value.toLowerCase();
            const items = document.querySelectorAll('.emp-item');

            items.forEach(item => {
                const name = item.querySelector('.emp-name').textContent.toLowerCase();
                const dept = item.querySelector('.emp-dept').textContent.toLowerCase();
                if (name.includes(input) || dept.includes(input)) {
                    item.style.display = "flex";
                } else {
                    item.style.display = "none";
                }
            });
        }

        // 공지사항 탭 전환 UI
        function switchNoticeTab(type) {
            const tabs = ['all', 'compliance', 'hr'];
            tabs.forEach(tab => {
                const btn = document.getElementById(`tab-${tab}`);
                if (tab === type) {
                    btn.className = "py-3 text-sm font-bold border-b-2 border-kb-yellow text-kb-charcoal";
                } else {
                    btn.className = "py-3 text-sm font-semibold text-kb-grayText hover:text-kb-charcoal border-b-2 border-transparent";
                }
            });
        }

        // 결재 모달 열기 및 닫기
        function openApprovalModal(docId) {
            document.getElementById('modalDocId').textContent = docId;
            document.getElementById('approvalModal').classList.remove('hidden');
            document.getElementById('approvalModal').classList.add('flex');
        }

        function closeApprovalModal() {
            document.getElementById('approvalModal').classList.add('hidden');
            document.getElementById('approvalModal').classList.remove('flex');
            document.getElementById('approvalComment').value = '';
        }

        function processApproval(action) {
            const comment = document.getElementById('approvalComment').value;
            if (action === 'approve') {
                alert('전자서명이 정상 완료되어 승인 처리되었습니다.');
            } else {
                alert('해당 안건이 반려 처리되었습니다.');
            }
            closeApprovalModal();
        }

        // 키보드 단축키 지원 (⌘K / Ctrl+K 로 검색창 포커스)
        window.addEventListener('keydown', (e) => {
            if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
                e.preventDefault();
                document.getElementById('globalSearch').focus();
            }
